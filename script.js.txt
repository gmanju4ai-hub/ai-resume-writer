function generateResume() {

  const name = document.getElementById("name").value;
  const email = document.getElementById("email").value;
  const education = document.getElementById("education").value;
  const skills = document.getElementById("skills").value;
  const experience = document.getElementById("experience").value;

  const resume = `
========================
        RESUME
========================

NAME: ${name}
EMAIL: ${email}

SUMMARY:
Motivated and dedicated student seeking opportunities to apply skills and grow professionally.

EDUCATION:
${education}

SKILLS:
${skills}

EXPERIENCE / PROJECTS:
${experience || "No experience provided"}

========================
Generated by ResumeAI
========================
`;

  document.getElementById("output").innerText = resume;
}

/* Simple PDF download (basic version) */
function downloadResume() {
  const text = document.getElementById("output").innerText;
  const blob = new Blob([text], { type: "text/plain" });
  const link = document.createElement("a");

  link.href = URL.createObjectURL(blob);
  link.download = "resume.txt";
  link.click();
}
